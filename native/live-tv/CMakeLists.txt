cmake_minimum_required(VERSION 3.25)
project(koko_tv LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(koko_tv SHARED bridge.cpp)
if(WIN32)
    target_compile_definitions(koko_tv PRIVATE _TSCOREDLL_USE _TSDUCKDLL_USE)
    target_include_directories(koko_tv PRIVATE "${TSDUCK_ROOT}/include/tsduck" "${TSDUCK_ROOT}/include/tscore")
    if(CMAKE_GENERATOR_PLATFORM MATCHES "ARM64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|aarch64")
        set(TSDUCK_ARCH Arm64)
    else()
        set(TSDUCK_ARCH Win64)
    endif()
    target_link_libraries(koko_tv PRIVATE "${TSDUCK_ROOT}/lib/Release-${TSDUCK_ARCH}/tsduck.lib"
                                          "${TSDUCK_ROOT}/lib/Release-${TSDUCK_ARCH}/tscore.lib")
    target_compile_options(koko_tv PRIVATE /W4 /EHsc /utf-8)
else()
    if(TSDUCK_SOURCE)
        file(GLOB_RECURSE TSDUCK_HEADERS "${TSDUCK_SOURCE}/src/libtscore/*.h" "${TSDUCK_SOURCE}/src/libtsduck/*.h")
        set(TSDUCK_INCLUDES "${TSDUCK_RUNTIME}/include")
        foreach(header IN LISTS TSDUCK_HEADERS)
            if(header MATCHES "/private/")
                continue()
            endif()
            if(APPLE AND header MATCHES "/(linux|windows|bsd|freebsd|netbsd|openbsd|dragonflybsd)/")
                continue()
            endif()
            if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND header MATCHES
                                                      "/(mac|windows|bsd|freebsd|netbsd|openbsd|dragonflybsd)/")
                continue()
            endif()
            get_filename_component(directory "${header}" DIRECTORY)
            list(APPEND TSDUCK_INCLUDES "${directory}")
        endforeach()
        list(REMOVE_DUPLICATES TSDUCK_INCLUDES)
        target_include_directories(koko_tv PRIVATE ${TSDUCK_INCLUDES})
        target_link_directories(koko_tv PRIVATE "${TSDUCK_RUNTIME}")
        target_link_libraries(koko_tv PRIVATE tsduck tscore)
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(TSDUCK REQUIRED IMPORTED_TARGET tsduck)
        target_link_libraries(koko_tv PRIVATE PkgConfig::TSDUCK)
    endif()
    set_target_properties(koko_tv PROPERTIES BUILD_WITH_INSTALL_RPATH ON)
    set_target_properties(koko_tv PROPERTIES INSTALL_RPATH "$ORIGIN;@loader_path")
endif()
install(TARGETS koko_tv DESTINATION .)
